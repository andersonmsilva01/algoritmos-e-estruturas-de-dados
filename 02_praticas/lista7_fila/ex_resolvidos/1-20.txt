#include <stdio.h>
#include <stdlib.h>

#define CAPACIDADE 100

typedef struct fila {
    int dados[CAPACIDADE];
    int primeiro;
    int ultimo;
    int quantidade;
} Fila;

Fila *cria(void) {
    Fila *f = malloc(sizeof *f);

    if (f != NULL) {
        f->primeiro = 0;
        f->ultimo = 0;
        f->quantidade = 0;
    }

    return f;
}

void destroi(Fila *f) {
    free(f);
}

int esta_vazia(const Fila *f) {
    return f->quantidade == 0;
}

int esta_cheia(const Fila *f) {
    return f->quantidade == CAPACIDADE;
}

int tamanho(const Fila *f) {
    return f->quantidade;
}

int enfilera(Fila *f, int valor) {
    if (esta_cheia(f)) {
        return 0;
    }

    f->dados[f->ultimo] = valor;
    f->ultimo = (f->ultimo + 1) % CAPACIDADE;
    f->quantidade++;
    return 1;
}

int desenfilera(Fila *f, int *valor) {
    if (esta_vazia(f)) {
        return 0;
    }

    *valor = f->dados[f->primeiro];
    f->primeiro = (f->primeiro + 1) % CAPACIDADE;
    f->quantidade--;
    return 1;
}

int frente(const Fila *f, int *valor) {
    if (esta_vazia(f)) {
        return 0;
    }

    *valor = f->dados[f->primeiro];
    return 1;
}

        return 0;
    }

    int sao_iguais = 1;

        int valor_a;
        int valor_b;

        desenfilera(a, &valor_a);
        desenfilera(b, &valor_b);
        enfilera(a, valor_a);
        enfilera(b, valor_b);

            sao_iguais = 0;
        }
    }

    return sao_iguais;

Fila *intercala(Fila *a, Fila *b) {
    if (tamanho(a) + tamanho(b) > CAPACIDADE) {
        return NULL;
    }

    Fila *resultado = cria();
    if (resultado == NULL) {
        return NULL;
    }

    int quantidade_a = tamanho(a);
    int quantidade_b = tamanho(b);
    int maior_quantidade = quantidade_a;

    if (quantidade_b > maior_quantidade) {
        maior_quantidade = quantidade_b;
    }

    for (int i = 0; i < maior_quantidade; i++) {
        int valor;

        if (i < quantidade_a) {
            desenfilera(a, &valor);
            enfilera(a, valor);
            enfilera(resultado, valor);
        }

        if (i < quantidade_b) {
            desenfilera(b, &valor);
            enfilera(b, valor);
            enfilera(resultado, valor);
        }
    }

    return resultado;
}

As filas a e b terminam intactas. A funcao retorna NULL se a soma dos tamanhos
ultrapassar a capacidade da fila resultante.

    if (auxiliar == NULL) {
        return;
    }

    int valor;
    while (desenfilera(f, &valor)) {
        if (valor != procurado) {

        }

        enfilera(f, valor);
    }

    destroi(auxiliar);
}

    f->inicio = 0;

    f->quantidade = 0;

int enfileirar_string(FilaStrings *f, const char texto[]) {
    if (f->quantidade == MAX_ITENS) {
        return 0;
    }

    strcpy(f->dados[f->fim], texto);
    f->fim = (f->fim + 1) % MAX_ITENS;
    f->quantidade++;
    return 1;

    if (f->quantidade == 0) {
        return 0;

    strcpy(texto, f->dados[f->inicio]);
    f->inicio = (f->inicio + 1) % MAX_ITENS;
    f->quantidade--;
    return 1;
}

int main(void) {
    FilaStrings fila;
    int n;

    printf("N (1 a 100): ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Valor invalido.\n");

    inicializar(&fila);
    enfileirar_string(&fila, "1");

EXERCICIO 14 - Simular escalonamento round-robin

#include <stdio.h>

#define MAX_PROCESSOS 100

typedef struct {
    int id;
    int tempo_restante;
} Processo;

typedef struct {
    Processo dados[MAX_PROCESSOS];
    int inicio;
    int fim;
    int quantidade;
} FilaProcessos;

void inicializar(FilaProcessos *f) {
    f->inicio = 0;
    f->fim = 0;
    f->quantidade = 0;
}

int enfileirar_processo(FilaProcessos *f, Processo processo) {
    if (f->quantidade == MAX_PROCESSOS) {
        return 0;
    }

    f->dados[f->fim] = processo;
    f->fim = (f->fim + 1) % MAX_PROCESSOS;
    f->quantidade++;
    return 1;
}

int desenfileirar_processo(FilaProcessos *f, Processo *processo) {
    if (f->quantidade == 0) {
        return 0;
    }

    *processo = f->dados[f->inicio];
    f->inicio = (f->inicio + 1) % MAX_PROCESSOS;
    f->quantidade--;
    return 1;
}

int main(void) {
    FilaProcessos fila;
    int quantidade;
    int quantum;
    int tempo_total = 0;

    inicializar(&fila);

    printf("Quantidade de processos: ");
    scanf("%d", &quantidade);
    printf("Quantum: ");
    scanf("%d", &quantum);

    if (quantidade < 1 || quantidade > MAX_PROCESSOS || quantum < 1) {
        printf("Valores invalidos.\n");
        return 1;
    }

    for (int i = 0; i < quantidade; i++) {
        Processo processo;
        processo.id = i + 1;
        printf("Tempo total do processo P%d: ", processo.id);
        scanf("%d", &processo.tempo_restante);
        enfileirar_processo(&fila, processo);
    }

    printf("Ordem de termino:\n");
    while (fila.quantidade > 0) {
        Processo processo;
        desenfileirar_processo(&fila, &processo);

        if (processo.tempo_restante <= quantum) {
            tempo_total += processo.tempo_restante;
            processo.tempo_restante = 0;
            printf("P%d terminou no tempo %d.\n", processo.id, tempo_total);
        } else {
            processo.tempo_restante -= quantum;
            tempo_total += quantum;
            enfileirar_processo(&fila, processo);
        }
    }

    return 0;
}

EXERCICIO 16 - Implementar uma fila usando duas pilhas
===============================================================================

#include <stdio.h>

#define CAPACIDADE 100

typedef struct {
    int dados[CAPACIDADE];
    int topo;
} Pilha;

typedef struct {
    Pilha entrada;
    Pilha saida;
} FilaDuasPilhas;

void pilha_inicializar(Pilha *p) {
    p->topo = 0;
}

int pilha_empilhar(Pilha *p, int valor) {
    if (p->topo == CAPACIDADE) {
        return 0;
    }

    p->dados[p->topo] = valor;
    p->topo++;
    return 1;
}

int pilha_desempilhar(Pilha *p, int *valor) {
    if (p->topo == 0) {
        return 0;
    }

    p->topo--;
    *valor = p->dados[p->topo];
    return 1;
}

void fila_inicializar(FilaDuasPilhas *f) {
    pilha_inicializar(&f->entrada);
    pilha_inicializar(&f->saida);
}

int fila_tamanho(const FilaDuasPilhas *f) {
    return f->entrada.topo + f->saida.topo;
}

int enfilera(FilaDuasPilhas *f, int valor) {
    if (fila_tamanho(f) == CAPACIDADE) {
        return 0;
    }

    return pilha_empilhar(&f->entrada, valor);
}

int desenfilera(FilaDuasPilhas *f, int *valor) {
    if (f->saida.topo == 0) {
        int temporario;

        while (pilha_desempilhar(&f->entrada, &temporario)) {
            pilha_empilhar(&f->saida, temporario);
        }
    }

    return pilha_desempilhar(&f->saida, valor);
}

/*
Enfileirar custa O(1). Desenfileirar pode custar O(n) durante uma transferencia,
mas cada elemento e transferido somente uma vez; portanto, o custo amortizado
de cada operacao e O(1).
*/

EXERCICIO 17 - Implementar uma fila com ponteiros

#include <stdio.h>
#include <stdlib.h>

typedef struct no {
    int valor;
    struct no *proximo;
} No;

typedef struct fila {
    No *primeiro;
    No *ultimo;
    int quantidade;
} Fila;

Fila *cria(void) {
    Fila *f = malloc(sizeof *f);

    if (f != NULL) {
        f->primeiro = NULL;
        f->ultimo = NULL;
        f->quantidade = 0;
    }

    return f;
}

int enfilera(Fila *f, int valor) {
    No *novo = malloc(sizeof *novo);

    if (novo == NULL) {
        return 0;
    }

    novo->valor = valor;
    novo->proximo = NULL;

    if (f->ultimo == NULL) {
        f->primeiro = novo;
    } else {
        f->ultimo->proximo = novo;
    }

    f->ultimo = novo;
    f->quantidade++;
    return 1;
}

int desenfilera(Fila *f, int *valor) {
    if (f->primeiro == NULL) {
        return 0;
    }

    No *removido = f->primeiro;
    *valor = removido->valor;
    f->primeiro = removido->proximo;

    if (f->primeiro == NULL) {
        f->ultimo = NULL;
    }

    free(removido);
    f->quantidade--;
    return 1;
}

int tamanho(const Fila *f) {
    return f->quantidade;
}

void destroi(Fila *f) {
    int valor;
    while (desenfilera(f, &valor)) {
    }
    free(f);
}

/*
O ponteiro ultimo permite inserir diretamente no fim em O(1). Sem ele, seria
necessario percorrer todos os nos. Nao existe capacidade fixa; a fila so deixa
de crescer se malloc falhar.
*/

#include <stdio.h>

#define CAPACIDADE 100

typedef struct {
    int dados[CAPACIDADE];
    int inicio;
    int quantidade;
} Deque;

void inicializar(Deque *d) {
    d->inicio = 0;
    d->quantidade = 0;
}

int insere_inicio(Deque *d, int valor) {
    if (d->quantidade == CAPACIDADE) {
        return 0;
    }

    d->inicio = (d->inicio - 1 + CAPACIDADE) % CAPACIDADE;
    d->dados[d->inicio] = valor;
    d->quantidade++;
    return 1;
}

int insere_fim(Deque *d, int valor) {
    if (d->quantidade == CAPACIDADE) {
        return 0;
    }

    int fim = (d->inicio + d->quantidade) % CAPACIDADE;
    d->dados[fim] = valor;
    d->quantidade++;
    return 1;
}

int remove_inicio(Deque *d, int *valor) {
    if (d->quantidade == 0) {
        return 0;
    }

    *valor = d->dados[d->inicio];
    d->inicio = (d->inicio + 1) % CAPACIDADE;
    d->quantidade--;
    return 1;
}

int remove_fim(Deque *d, int *valor) {
    if (d->quantidade == 0) {
        return 0;
    }

    int fim = (d->inicio + d->quantidade - 1) % CAPACIDADE;
    *valor = d->dados[fim];
    d->quantidade--;
    return 1;
}

    return 1;
}

int desenfilera(FilaPrioridade *f, int *valor) {
    if (f->quantidade == 0) {
        return 0;
    }

    int indice_maior = 0;

    for (int i = 1; i < f->quantidade; i++) {
        if (f->itens[i].prioridade > f->itens[indice_maior].prioridade) {
            indice_maior = i;
        }
    }

    *valor = f->itens[indice_maior].valor;

    for (int i = indice_maior; i < f->quantidade - 1; i++) {
        f->itens[i] = f->itens[i + 1];
    }

    f->quantidade--;
    return 1;
}

A insercao e O(1) e a remocao e O(n). Como a busca usa apenas >, em caso de
empate o primeiro item de maior prioridade e removido, preservando a ordem de
chegada. Uma heap reduziria a remocao para O(log n), mas exigiria um contador de
chegada para manter a estabilidade nos empates.
