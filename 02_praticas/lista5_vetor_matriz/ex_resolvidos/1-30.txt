    int maior = vetor[0];
    int menor = vetor[0];

    for (int i = 1; i < 12; i++) {
        if (vetor[i] > maior) {
            maior = vetor[i];
        }
        if (vetor[i] < menor) {
            menor = vetor[i];
        }
    }

    printf("Maior: %d\n", maior);
    printf("Menor: %d\n", menor);

    return 0;
}

EXERCICIO 11 - Ordenar um vetor em ordem crescente
===============================================================================

#include <stdio.h>

int main(void) {
    int vetor[8];

    for (int i = 0; i < 8; i++) {
        printf("Digite o valor %d: ", i + 1);
        scanf("%d", &vetor[i]);
    }

    for (int fim = 7; fim > 0; fim--) {
        for (int i = 0; i < fim; i++) {
            if (vetor[i] > vetor[i + 1]) {
                int temporario = vetor[i];
                vetor[i] = vetor[i + 1];
                vetor[i + 1] = temporario;
            }
        }
    }

    printf("Ordem crescente: ");
    for (int i = 0; i < 8; i++) {
        printf("%d ", vetor[i]);
    }
    printf("\n");

    return 0;
}

EXERCICIO 15 - Verificar se existem valores repetidos no vetor
===============================================================================

#include <stdio.h>

int main(void) {
    int vetor[10];
    int repetido = 0;

    for (int i = 0; i < 10; i++) {
        printf("Digite o valor %d: ", i + 1);
        scanf("%d", &vetor[i]);
    }

    for (int i = 0; i < 9 && !repetido; i++) {
        for (int j = i + 1; j < 10; j++) {
            if (vetor[i] == vetor[j]) {
                repetido = 1;
                break;
            }
        }
    }

    if (repetido) {
        printf("Existem valores repetidos.\n");
    } else {
        printf("Nao existem valores repetidos.\n");
    }

    return 0;
}

EXERCICIO 26 - Verificar se uma matriz 3 x 3 e identidade
===============================================================================

#include <stdio.h>

int main(void) {
    int matriz[3][3];
    int identidade = 1;

    for (int linha = 0; linha < 3; linha++) {
        for (int coluna = 0; coluna < 3; coluna++) {
            printf("Elemento [%d][%d]: ", linha, coluna);
            scanf("%d", &matriz[linha][coluna]);
        }
    }

    for (int linha = 0; linha < 3; linha++) {
        for (int coluna = 0; coluna < 3; coluna++) {
            if (linha == coluna && matriz[linha][coluna] != 1) {
                identidade = 0;
            }
            if (linha != coluna && matriz[linha][coluna] != 0) {
                identidade = 0;
            }
        }
    }

    if (identidade) {
        printf("A matriz e identidade.\n");
    } else {
        printf("A matriz nao e identidade.\n");
    }

    return 0;
}

EXERCICIO 29 - Multiplicar uma matriz 2 x 3 por uma matriz 3 x 2
===============================================================================

#include <stdio.h>

int main(void) {
    int a[2][3];
    int b[3][2];
    int produto[2][2] = {0};

    printf("Matriz 2 x 3:\n");
    for (int linha = 0; linha < 2; linha++) {
        for (int coluna = 0; coluna < 3; coluna++) {
            scanf("%d", &a[linha][coluna]);
        }
    }

    printf("Matriz 3 x 2:\n");
    for (int linha = 0; linha < 3; linha++) {
        for (int coluna = 0; coluna < 2; coluna++) {
            scanf("%d", &b[linha][coluna]);
        }
    }

    for (int linha = 0; linha < 2; linha++) {
        for (int coluna = 0; coluna < 2; coluna++) {
            for (int k = 0; k < 3; k++) {
                produto[linha][coluna] += a[linha][k] * b[k][coluna];
            }
        }
    }

    printf("Produto:\n");
    for (int linha = 0; linha < 2; linha++) {
        for (int coluna = 0; coluna < 2; coluna++) {
            printf("%d ", produto[linha][coluna]);
        }
        printf("\n");
    }

    return 0;
}

EXERCICIO 30 - Encontrar a posicao do maior elemento de uma matriz 5 x 5
===============================================================================

#include <stdio.h>

int main(void) {
    int matriz[5][5];

    for (int linha = 0; linha < 5; linha++) {
        for (int coluna = 0; coluna < 5; coluna++) {
            printf("Elemento [%d][%d]: ", linha, coluna);
            scanf("%d", &matriz[linha][coluna]);
        }
    }

    int maior = matriz[0][0];
    int linha_maior = 0;
    int coluna_maior = 0;

    for (int linha = 0; linha < 5; linha++) {
        for (int coluna = 0; coluna < 5; coluna++) {
            if (matriz[linha][coluna] > maior) {
                maior = matriz[linha][coluna];
                linha_maior = linha;
                coluna_maior = coluna;
            }
        }
    }

    printf("Maior elemento: %d\n", maior);
    printf("Posicao: linha %d, coluna %d\n", linha_maior, coluna_maior);

    return 0;
}
